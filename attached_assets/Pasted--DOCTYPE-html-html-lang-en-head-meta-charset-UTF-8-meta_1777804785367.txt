  // ========== CONFIGURATION ==========
  // 🔴 CHANGE THIS TO YOUR REAL WHATSAPP NUMBER (e.g., "+15871234567")
  const OWNER_WHATSAPP = "+15871234567";

  // 🔴 EMAILJS (optional – get free keys from emailjs.com)
  // If you don't set these, forms will show success but won't send emails.
  const EMAILJS_PUBLIC_KEY = "YOUR_PUBLIC_KEY";
  const EMAILJS_SERVICE_ID = "service_xxxx";
  const EMAILJS_TEMPLATE_ASSESS = "template_xxx";
  const EMAILJS_TEMPLATE_CONTACT = "template_yyy";

  let emailjsReady = false;
  if (EMAILJS_PUBLIC_KEY && EMAILJS_PUBLIC_KEY !== "YOUR_PUBLIC_KEY") {
    emailjs.init(EMAILJS_PUBLIC_KEY);
    emailjsReady = true;
  }

  function updateWhatsAppLinks() {
    const cleanNumber = OWNER_WHATSAPP.replace(/\D/g, '');
    const waUrl = `https://wa.me/${cleanNumber}?text=${encodeURIComponent("Hello CanadaReady! I need help.")}`;
    const ids = ['whatsappFloat', 'heroWhatsappBtn', 'contactWhatsappLink'];
    ids.forEach(id => {
      const el = document.getElementById(id);
      if (el) el.href = waUrl;
    });
    ['contactNumberDisplay', 'footerNumber'].forEach(id => {
      const el = document.getElementById(id);
      if (el) el.innerText = OWNER_WHATSAPP;
    });
  }
  updateWhatsAppLinks();

  function setLang(lang) {
    document.body.classList.remove('ar');
    if (lang === 'ar') document.body.classList.add('ar');
    localStorage.setItem('lang', lang);
    document.getElementById('btnEn').classList.toggle('active', lang === 'en');
    document.getElementById('btnAr').classList.toggle('active', lang === 'ar');
  }

  function navigate(pageId) {
    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    document.getElementById(`page-${pageId}`).classList.add('active');
    window.scrollTo(0, 0);
  }

  const savedLang = localStorage.getItem('lang');
  if (savedLang) setLang(savedLang);

  // Assessment form submission
  document.getElementById('assessmentForm').addEventListener('submit', async function(e) {
    e.preventDefault();
    const formData = {
      name: document.getElementById('assessName').value,
      whatsapp: document.getElementById('assessWhatsapp').value,
      email: document.getElementById('assessEmail').value,
      level: document.getElementById('assessLevel').value,
      city: document.getElementById('assessCity').value,
      goals: document.getElementById('assessGoals').value
    };
    if (emailjsReady && EMAILJS_SERVICE_ID && EMAILJS_TEMPLATE_ASSESS) {
      try {
        await emailjs.send(EMAILJS_SERVICE_ID, EMAILJS_TEMPLATE_ASSESS, formData);
      } catch(err) { console.warn('EmailJS error:', err); }
    }
    document.getElementById('assessmentForm').style.display = 'none';
    document.getElementById('formSuccess').style.display = 'block';
  });

    };

    }
    document.getElementById('contactForm').style.display = 'none';

  });
